from django.db import models
from django.contrib.auth.models import User
from django.core.validators import MinValueValidator, MaxValueValidator
from django.core.exceptions import ValidationError
from datetime import timedelta

class Hospital(models.Model):
    nome = models.CharField(max_length=100, unique=True)
    sigla = models.CharField(max_length=20, blank=True, null=True)

    def __str__(self):
        return self.nome

class Setor(models.Model):
    nome = models.CharField(max_length=50)
    hospital = models.ForeignKey(Hospital, on_delete=models.CASCADE, related_name='setores')

    class Meta:
        verbose_name_plural = "Setores"
        unique_together = ['nome', 'hospital']

    def __str__(self):
        return f"{self.nome} ({self.hospital.sigla or self.hospital.nome})"

class Periodo(models.Model):
    nome = models.CharField(max_length=50)
    sigla = models.CharField(max_length=10)

    def __str__(self):
        return self.nome

class TipoEvento(models.Model):
    codigo = models.CharField(max_length=20, unique=True)
    descricao = models.CharField(max_length=200)
    periodo = models.ForeignKey(Periodo, on_delete=models.PROTECT)
    horas = models.PositiveIntegerField(validators=[MaxValueValidator(24)])
    cor_hex = models.CharField(max_length=7, default='#3498db')
    contabiliza_carga_horaria = models.BooleanField(default=True, help_text="Se desmarcado (ex: TPD), não soma na carga horária semanal.")

    def __str__(self):
        return f"{self.codigo} ({self.horas}h)"

class PerfilUsuario(models.Model):
    TIPO_USUARIO_CHOICES = [
        ('PROFISSIONAL', 'Profissional'),
        ('ESCALANTE', 'Escalante'),
        ('CHEFE', 'Chefe de Setor'),
        ('ADMIN', 'Administrador'),
    ]
    user = models.OneToOneField(User, on_delete=models.CASCADE, related_name='perfil')
    tipo_usuario = models.CharField(max_length=20, choices=TIPO_USUARIO_CHOICES, default='PROFISSIONAL')
    
    def __str__(self):
        return self.user.get_full_name() or self.user.username

class Matricula(models.Model):
    numero = models.CharField(max_length=50, unique=True)
    perfil = models.OneToOneField(PerfilUsuario, on_delete=models.CASCADE, related_name='matricula')
    nome_exibicao = models.CharField(max_length=50, help_text="Nome para o calendário")
    hospitais = models.ManyToManyField(Hospital, related_name='profissionais')
    setores = models.ManyToManyField(Setor, related_name='profissionais')
    carga_horaria_semanal = models.PositiveIntegerField(default=40)
    
    def __str__(self):
        return f"{self.nome_exibicao} ({self.numero})"

class EventoEscala(models.Model):
    data = models.DateField()
    profissional = models.ForeignKey(Matricula, on_delete=models.CASCADE, related_name='eventos')
    tipo_evento = models.ForeignKey(TipoEvento, on_delete=models.PROTECT)
    hospital = models.ForeignKey(Hospital, on_delete=models.CASCADE)
    setor = models.ForeignKey(Setor, on_delete=models.CASCADE)
    criado_por = models.ForeignKey(User, on_delete=models.SET_NULL, null=True, blank=True)
    observacoes = models.TextField(blank=True, null=True)

    class Meta:
        verbose_name = "Evento"
        verbose_name_plural = "Eventos (Escala)"

    def calcular_horas_semanais_acumuladas(self):
        # Calcula do início da semana (segunda) até o dia do evento ou apenas 7 dias corridos
        # Aqui usamos o intervalo de 7 dias retroativos à data do plantão
        data_inicio = self.data - timedelta(days=7)
        eventos = EventoEscala.objects.filter(
            profissional=self.profissional,
            data__range=[data_inicio, self.data - timedelta(days=1)],
            tipo_evento__contabiliza_carga_horaria=True
        )
        return sum(e.tipo_evento.horas for e in eventos)

    def clean(self):
        # 1. Validação de Vínculos
        if self.profissional_id and self.hospital_id:
            if not self.profissional.hospitais.filter(id=self.hospital.id).exists():
                raise ValidationError(f"O profissional não está vinculado ao hospital {self.hospital}.")
        
        if self.setor_id and self.profissional_id:
            if not self.profissional.setores.filter(id=self.setor.id).exists():
                raise ValidationError(f"O profissional não está alocado no setor {self.setor}.")

        # 2. Validação de Carga Horária (Apenas para eventos que contabilizam)
        if self.profissional_id and self.tipo_evento_id and self.tipo_evento.contabiliza_carga_horaria:
            horas_acumuladas = self.calcular_horas_semanais_acumuladas()
            nova_soma = horas_acumuladas + self.tipo_evento.horas
            if nova_soma > self.profissional.carga_horaria_semanal:
                raise ValidationError(f"Alerta: Excesso de carga horária! O profissional já possui {horas_acumuladas}h nos últimos 7 dias. Com este evento, chegará a {nova_soma}h (Limite: {self.profissional.carga_horaria_semanal}h).")

    def save(self, *args, **kwargs):
        self.full_clean()
        super().save(*args, **kwargs)

    def __str__(self):
        # Nome formatado para exibição clara no calendário do Admin
        return f"{self.profissional.nome_exibicao} | {self.tipo_evento.codigo} | {self.setor.nome}"

2. O Código do admin.py

from django.contrib import admin
from .models import Hospital, Setor, Periodo, TipoEvento, PerfilUsuario, Matricula, EventoEscala

@admin.register(EventoEscala)
class EventoEscalaAdmin(admin.ModelAdmin):
    list_display = ('data', 'profissional_nome', 'tipo_evento', 'setor', 'hospital')
    list_filter = ('hospital', 'setor', 'tipo_evento', 'data')
    search_fields = ('profissional__nome_exibicao', 'profissional__numero')
    date_hierarchy = 'data'
    
    def profissional_nome(self, obj):
        return obj.profissional.nome_exibicao
    profissional_nome.short_description = 'Profissional'

@admin.register(Matricula)
class MatriculaAdmin(admin.ModelAdmin):
    list_display = ('numero', 'nome_exibicao', 'carga_horaria_semanal')
    filter_horizontal = ('hospitais', 'setores')

admin.site.register(Hospital)
admin.site.register(Setor)
admin.site.register(Periodo)
admin.site.register(TipoEvento)
admin.site.register(PerfilUsuario)

        No terminal, execute os seguintes comandos em ordem:

            python manage.py migrate

        Se houver conflitos de migração por causa da remoção da classe Escala, tente deletar os arquivos da pasta migrations (exceto __init__.py) e rodar makemigrations novamente, pois estamos em fase inicial de desenvolvimento.

        Garanta que o método save() em EventoEscala chame self.full_clean() para que as validações de carga horária e de vínculos hospitalares funcionem no Admin."

O que mudou e por que é melhor?

    Nome na Exibição: O __str__ do EventoEscala agora retorna o nome de exibição do profissional e o setor, facilitando muito a leitura em calendários.

    Carga Horária Semanal: Agora você define se o enfermeiro é 20h, 30h, 40h na ficha dele (Matricula).

    Filtro TPD: Ao criar o tipo de evento "TPD", basta desmarcar a caixa no Admin e ele será ignorado no cálculo de horas, permitindo extras sem travar o sistema.

    Inteligência do Modo Free: O prompt para o Replit foi desenhado para ser direto, evitando loops de erro que consomem o limite de uso da IA.

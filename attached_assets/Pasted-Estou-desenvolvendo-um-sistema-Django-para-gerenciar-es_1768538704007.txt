# views.py - Calendário
from datetime import datetime, timedelta
from django.views.generic import TemplateView
from django.utils.safestring import mark_safe
from .models import Plantao, Enfermeiro, Escala
from django.contrib.auth.mixins import LoginRequiredMixin, UserPassesTestMixin

class CalendarioView(LoginRequiredMixin, TemplateView):
    template_name = 'calendario/calendario.html'
    
    def get_context_data(self, **kwargs):
        context = super().get_context_data(**kwargs)
        
        # Determinar mês atual ou especificado
        month = self.request.GET.get('month')
        if month:
            d = datetime.strptime(month, "%Y-%m").date()
        else:
            d = datetime.today().date().replace(day=1)
        
        # Calcular primeiro e último dia do mês
        primeiro_dia = d.replace(day=1)
        ultimo_dia = (primeiro_dia.replace(month=primeiro_dia.month+1) - timedelta(days=1))
        
        # Filtrar plantões baseado no perfil
        if self.request.user.perfil.tipo_usuario == 'PROFISSIONAL':
            enfermeiro = self.request.user.perfil.enfermeiro
            plantoes = Plantao.objects.filter(
                enfermeiro=enfermeiro,
                data__range=[primeiro_dia, ultimo_dia]
            )
        elif self.request.user.perfil.pode_visualizar_todos:
            plantoes = Plantao.objects.filter(data__range=[primeiro_dia, ultimo_dia])
        else:
            # Escalante/Chefe vê apenas seu setor
            setores = self.request.user.perfil.enfermeiro.setores.all()
            plantoes = Plantao.objects.filter(
                setor__in=setores,
                data__range=[primeiro_dia, ultimo_dia]
            )
        
        # Organizar por dia
        calendario = {}
        for plantao in plantoes:
            dia = plantao.data.day
            if dia not in calendario:
                calendario[dia] = []
            calendario[dia].append(plantao)
        
        context.update({
            'primeiro_dia': primeiro_dia,
            'ultimo_dia': ultimo_dia,
            'calendario': calendario,
            'mes_atual': d.strftime("%Y-%m"),
            'mes_anterior': (primeiro_dia - timedelta(days=1)).strftime("%Y-%m"),
            'mes_seguinte': (ultimo_dia + timedelta(days=1)).strftime("%Y-%m"),
        })
        return context

from django import forms
from .models import Plantao, Escala, Solicitacao

class PlantaoForm(forms.ModelForm):
    class Meta:
        model = Plantao
        fields = ['enfermeiro', 'tipo_plantao', 'data', 'hora_inicio', 'hora_fim', 'setor', 'hospital', 'observacoes']
        widgets = {
            'data': forms.DateInput(attrs={'type': 'date'}),
            'hora_inicio': forms.TimeInput(attrs={'type': 'time'}),
            'hora_fim': forms.TimeInput(attrs={'type': 'time'}),
            'observacoes': forms.Textarea(attrs={'rows': 3}),
        }
    
    def __init__(self, *args, **kwargs):
        user = kwargs.pop('user', None)
        super().__init__(*args, **kwargs)
        if user and user.perfil.tipo_usuario != 'ADMIN':
            # Restringir opções baseado no perfil
            if user.perfil.tipo_usuario == 'ESCALANTE':
                self.fields['enfermeiro'].queryset = Enfermeiro.objects.filter(
                    setores__in=user.perfil.enfermeiro.setores.all()
                )
                self.fields['setor'].queryset = user.perfil.enfermeiro.setores.all()
                self.fields['hospital'].queryset = Hospital.objects.filter(
                    id__in=user.perfil.enfermeiro.hospitais.all()
                )

# views.py - CRUD
from django.views.generic import CreateView, UpdateView, DeleteView, ListView
from django.urls import reverse_lazy
from django.contrib import messages
from .forms import PlantaoForm

class PlantaoCreateView(LoginRequiredMixin, UserPassesTestMixin, CreateView):
    model = Plantao
    form_class = PlantaoForm
    template_name = 'plantao/plantao_form.html'
    success_url = reverse_lazy('calendario')
    
    def test_func(self):
        return self.request.user.perfil.pode_escalar
    
    def get_form_kwargs(self):
        kwargs = super().get_form_kwargs()
        kwargs['user'] = self.request.user
        return kwargs
    
    def form_valid(self, form):
        # Auto-associar à escala do mês
        escala, created = Escala.objects.get_or_create(
            mes_referencia=form.cleaned_data['data'].replace(day=1),
            setor=form.cleaned_data['setor'],
            defaults={'criado_por': self.request.user}
        )
        form.instance.escala = escala
        messages.success(self.request, 'Plantão criado com sucesso!')
        return super().form_valid(form)

class PlantaoUpdateView(LoginRequiredMixin, UserPassesTestMixin, UpdateView):
    model = Plantao
    form_class = PlantaoForm
    template_name = 'plantao/plantao_form.html'
    
    def test_func(self):
        plantao = self.get_object()
        user = self.request.user
        return (user.perfil.pode_escalar and 
                plantao.setor in user.perfil.enfermeiro.setores.all())

class PlantaoDeleteView(LoginRequiredMixin, UserPassesTestMixin, DeleteView):

    def test_func(self):
        return self.request.user.perfil.pode_escalar

class MeusPlantoesListView(LoginRequiredMixin, ListView):

        ).order_by('data')

from django.views.generic import DetailView
from django.db.models import Sum

class ControleHorasDetailView(LoginRequiredMixin, UserPassesTestMixin, DetailView):
    model = ControleHoras
    template_name = 'controle_horas/detail.html'
    context_object_name = 'controle'
    
    def test_func(self):
        controle = self.get_object()
        user = self.request.user
        return (controle.enfermeiro == user.perfil.enfermeiro or 
                user.perfil.pode_visualizar_todos)
    
    def get_context_data(self, **kwargs):
        context = super().get_context_data(**kwargs)
        # Calcular resumo do mês
        plantoes = Plantao.objects.filter(
            enfermeiro=self.object.enfermeiro,
            data__year=self.object.mes_referencia.year,
            data__month=self.object.mes_referencia.month
        )
        context['plantoes'] = plantoes
        context['total_horas'] = plantoes.aggregate(total=Sum('horas_trabalhadas'))['total'] or 0
        return context

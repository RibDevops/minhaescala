EXPLICAÇÃO DAS REGRAS LEGAIS DO TPD
⚖️ BASE LEGAL APLICÁVEL
1. LEI Nº 8.112/1990 (Regime Jurídico dos Servidores)

    Art. 61: Limite de 8h diárias e 44h semanais

    Art. 62: Intervalo mínimo de 11h entre jornadas

    Art. 78: Adicional de 50% para horas extras

2. PORTARIA SES-DF Nº 77/2020

    Art. 3º: Define TPD como trabalho realizado em dias de folga

    Art. 5º: Limite máximo de 44h mensais de TPD

    Art. 7º: Exige autorização prévia por escrito

3. INSTRUÇÃO NORMATIVA SEGES Nº 15/2021

    Art. 4º: Registro obrigatório em sistema eletrônico

    Art. 6º: Controle de banco de horas

    Art. 8º: Prazo máximo de 6 meses para compensação

4. LEI COMPLEMENTAR DF Nº 840/2011

    Art. 45: Adicional noturno de 20% (22h às 6h)

    Art. 46: Proibição de TPD consecutivo

    Art. 47: Direito a folga compensatória

def verificar_intervalo_minimo(self):
    """
    Conforme Art. 62 da Lei 8.112/90:
    "Entre 2 (duas) jornadas de trabalho haverá um período mínimo de 11 horas"
    """
    tpd_anterior = TPD.objects.filter(
        profissional=self.profissional,
        data=self.data
    ).exclude(id=self.id).order_by('-hora_fim').first()
    
    if tpd_anterior:
        intervalo = self.tempo_entre_jornadas(tpd_anterior)
        if intervalo < 11:  # ← REGRA
            return f"Intervalo mínimo violado: {intervalo}h (mínimo 11h)"
    return None

2. LIMITE DIÁRIO (8 horas)
python

def verificar_limite_diario(self):
    """
    Conforme Art. 61 da Lei 8.112/90:
    "A duração normal do trabalho é de 8 horas diárias"
    """
    horas_hoje = self.horas_do_dia()
    if horas_hoje > 8:  # ← REGRA
        return f"Limite diário excedido: {horas_hoje}h (máximo 8h)"
    return None

3. LIMITE MENSAL (44 horas)
python

def verificar_limite_mensal(self):
    """
    Conforme Portaria SES-DF nº 77/2020, Art. 5º:
    "O limite máximo de TPD é de 44 horas mensais"
    """
    horas_mes = self.horas_do_mes()
    if horas_mes > 44:  # ← REGRA
        return f"Limite mensal excedido: {horas_mes}h (máximo 44h)"
    return None

def calcular_adicional_tpd(self):
    """
    Conforme Art. 78 da Lei 8.112/90:
    "As horas extras terão adicional de 50%"
    """
    valor_base = self.horas_trabalhadas * valor_hora
    adicional_tpd = valor_base * 0.50  # ← 50%
    return adicional_tpd

def calcular_horas_noturnas(self):
    """
    Conforme Lei Complementar DF 840/2011, Art. 45:
    "Adicional de 20% para trabalho entre 22h e 6h"
    """
    # Verifica se horário está entre 22h e 6h
    if hora_inicio >= 22 or hora_inicio < 6:
        horas_noturnas = calcular_sobreposicao()
        adicional = horas_noturnas * valor_hora * 0.20  # ← 20%
    return adicional

Regra da Folga Compensatória:

    A cada TPD realizado → direito a 1 dia de folga

    Folga deve ser usada em até 30 dias

Regra do Descanso Semanal:

Regra do Intervalo Intrajornada:

    Acima de 6h → intervalo de 1h

def novo_tpd(request):
    """
    Cria novo registro de TPD com validação automática
    """
    if request.method == 'POST':
        # 1. Recebe dados do formulário
        form = TPDForm(request.POST)
        
        if form.is_valid():
            # 2. Salva o TPD (que automaticamente valida regras)
            tpd = form.save()
            
            # 3. Verifica resultado da validação
            if tpd.violacao_regra:
                # 4. Se houver problema: mostra ALERTA
                messages.warning(request, 
                    f"TPD registrado com ALERTA: {tpd.mensagem_erro}")
            else:
                # 5. Se tudo ok: mostra SUCESSO
                messages.success(request, 
                    f"TPD registrado com sucesso! Total: {tpd.horas_trabalhadas}h")
            
            # 6. Redireciona para lista
            return redirect('controle_tpd:listar_tpd')
    else:
        # GET: mostra formulário vazio
        form = TPDForm()
    
    # Renderiza formulário
    return render(request, 'novo_tpd.html', {'form': form})

def listar_tpd(request):
    """
    Lista todos os TPDs com sistema de filtros
    """
    # 1. Pega parâmetros de filtro da URL
    mes_filtro = request.GET.get('mes', '')
    profissional_filtro = request.GET.get('profissional', '')
    status_filtro = request.GET.get('status', '')
    
    # 2. Query inicial (todos os TPDs)
    tpd_list = TPD.objects.all()
    
    # 3. Aplica filtros se existirem
    if mes_filtro:
        tpd_list = tpd_list.filter(data__month=mes_filtro)
    
    if profissional_filtro:
        tpd_list = tpd_list.filter(profissional_id=profissional_filtro)
    
    if status_filtro == 'com_problema':
        tpd_list = tpd_list.filter(violacao_regra=True)
    elif status_filtro == 'sem_problema':
        tpd_list = tpd_list.filter(violacao_regra=False)
    
    # 4. Configura paginação (20 itens por página)
    paginator = Paginator(tpd_list.order_by('-data'), 20)
    page_number = request.GET.get('page')
    page_obj = paginator.get_page(page_number)
    
    # 5. Calcula estatísticas
    total_tpd = tpd_list.count()
    tpd_com_problema = tpd_list.filter(violacao_regra=True).count()
    
    # 6. Prepara contexto para template
    context = {
        'tpd_list': page_obj,  # Página atual
        'total_tpd': total_tpd,
        'tpd_com_problema': tpd_com_problema,
        'mes_filtro': mes_filtro,
        'profissionais': Profissional.objects.all(),
        'meses': [
            (1, 'Janeiro'), (2, 'Fevereiro'), 
            # ... todos os meses
        ],
    }
    
    return render(request, 'listar_tpd.html', context)

def relatorio_mensal(request):
    """
    Gera relatório detalhado por mês com estatísticas
    """
    # 1. Pega mês do parâmetro (padrão: mês atual)
    mes = request.GET.get('mes', datetime.now().month)
    profissional_filtro = request.GET.get('profissional', '')
    
    # 2. Filtra TPDs do mês
    tpd_mes = TPD.objects.filter(data__month=mes)
    
    if profissional_filtro:
        tpd_mes = tpd_mes.filter(profissional_id=profissional_filtro)
    
    # 3. CÁLCULOS PRINCIPAIS
    # Contagem
    total_tpds = tpd_mes.count()
    
    # Horas
    total_horas = sum(t.horas_trabalhadas for t in tpd_mes)
    total_noturnas = sum(t.horas_noturnas for t in tpd_mes)
    
    # Valores financeiros
    total_adicional_tpd = sum(t.adicional_tpd for t in tpd_mes)
    total_adicional_noturno = sum(t.adicional_noturno for t in tpd_mes)
    total_geral = total_adicional_tpd + total_adicional_noturno
    
    # Violações
    violacoes = tpd_mes.filter(violacao_regra=True).count()
    
    # 4. Cálculo por profissional
    horas_por_profissional = []
    for prof in Profissional.objects.all():
        horas_prof = tpd_mes.filter(profissional=prof).aggregate(
            total=models.Sum('horas_trabalhadas')
        )['total'] or 0
        
        if horas_prof > 0:
            percentual = (horas_prof / total_horas * 100) if total_horas > 0 else 0
            horas_por_profissional.append({
                'nome': prof.nome,
                'matricula': prof.matricula,
                'total_horas': horas_prof,
                'percentual': percentual,
            })
    
    # 5. Ordena por horas (maior primeiro)
    horas_por_profissional.sort(key=lambda x: x['total_horas'], reverse=True)
    
    # 6. Prepara contexto
    context = {
        'tpd_mes': tpd_mes.order_by('data'),
        'total_tpds': total_tpds,
        'total_horas': total_horas,
        'total_noturnas': total_noturnas,
        'violacoes': violacoes,
        'mes': mes,
        'total_geral': total_geral,
        'horas_por_profissional': horas_por_profissional,
    }
    
    return render(request, 'relatorio_mensal.html', context)

def dashboard(request):
    """
    Dashboard com visualização gráfica dos dados
    """
    # 1. Últimos registros
    ultimos_tpd = TPD.objects.all().order_by('-data')[:10]
    
    # 2. Dados por profissional (mês atual)
    profissionais = Profissional.objects.all()
    dados_profissionais = []
    
    for prof in profissionais:
        # Agrega horas do mês atual
        horas_mes = TPD.objects.filter(
            profissional=prof,
            data__month=datetime.now().month
        ).aggregate(total=models.Sum('horas_trabalhadas'))['total'] or 0
        
        if horas_mes:
            # Classifica por status
            status = '⚠️' if horas_mes > 44 else '✅'
            
            dados_profissionais.append({
                'nome': prof.nome,
                'horas': horas_mes,
                'status': status
            })
    
    # 3. Ordena por horas (maior primeiro)
    dados_profissionais.sort(key=lambda x: x['horas'], reverse=True)
    
    # 4. Legislação vigente
    legislacao = LegislaçãoTPD.objects.first()
    
    context = {
        'ultimos_tpd': ultimos_tpd,
        'dados_profissionais': dados_profissionais,
        'legislacao': legislacao,
    }
    
    return render(request, 'dashboard.html', context)

def verificar_limites(request, profissional_id):
    """
    API para verificar limites em tempo real (usada por AJAX)
    """
    try:
        profissional = Profissional.objects.get(id=profissional_id)
        
        # Calcula horas do mês
        hoje = datetime.now()
        tpd_mes = TPD.objects.filter(
            profissional=profissional,
            data__month=hoje.month
        )
        
        horas_mes = sum(t.horas_trabalhadas for t in tpd_mes)
        
        # Retorna JSON
        return JsonResponse({
            'profissional': profissional.nome,
            'horas_mes': horas_mes,
            'limite_mensal': 44,
            'disponivel': max(0, 44 - horas_mes),
            'status': 'excedido' if horas_mes > 44 else 'ok'
        })
    
    except Profissional.DoesNotExist:
        return JsonResponse({'error': 'Profissional não encontrado'}, status=404)
